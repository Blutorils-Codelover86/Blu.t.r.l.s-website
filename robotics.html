Mastering pulse width modulation (PWM) by using Arduino Boards-

int ledPin = 9; // Pin connected to the LED
void setup() {
pinMode(ledPin, OUTPUT);
}
void loop() {
// Fade in (brightness from 0 to 255)
for (int i = 0; i <= 255; i++) {
analogWrite(ledPin, i);
delay(15);
}
// Fade out (brightness from 255 to 0)
for (int i = 255; i >= 0; i--) {
analogWrite(ledPin, i);
delay(15);
}
}
